        const match = f.match(/task-(\d+)/);
        return match ? parseInt(match[1], 10) : 0;
      });
  };
 
  const ids = [...getNumbers(activeDir), ...getNumbers(completedDir)];
  const maxId = Math.max(0, ...ids);
